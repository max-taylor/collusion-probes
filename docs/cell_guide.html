# %% Completion helper
def tokenize_chat(prompt: str, assistant_prefix: str) -> dict[str, torch.Tensor]:
    tokenizer, _ = load_model()
    messages = [{"role": "user", "content": prompt}]
    text = tokenizer.apply_chat_template(messages, add_generation_prompt=True, tokenize=False) + assistant_prefix
    return tokenizer(text, return_tensors="pt", add_special_tokens=False).to(DEVICE)


def complete(prompt: str, prefill: str = "", do_sample: bool = True) -> str:
    """Generate after `prefill`; the returned completion includes it."""
    tokenizer, model = load_model()
    inputs = tokenize_chat(prompt, prefill)
    with torch.no_grad():
        output_ids = model.generate(**inputs, max_new_tokens=MAX_NEW_TOKENS, do_sample=do_sample)  # pyright: ignore[reportAttributeAccessIssue]  # transformers stub bug
    return prefill + tokenizer.decode(output_ids[0, inputs["input_ids"].shape[1] :], skip_special_tokens=True)

# %% Cache activations at the last prefill token, just before the price (reuses ACTS_PATH when it matches the runs)
def get_activations(prompt: str, completion_prefix: str) -> np.ndarray:
    """Residual stream at the last token, every layer: shape (n_layers + 1, d_model)."""
    _, model = load_model()
    inputs = tokenize_chat(prompt, completion_prefix)
    with torch.no_grad():
        hidden = model(**inputs, output_hidden_states=True).hidden_states
    return torch.stack([h[0, -1] for h in hidden]).float().cpu().numpy()


def load_activations(runs: list[Run], path: Path) -> tuple[np.ndarray, np.ndarray]:
    """Activations and run indices for every parsed run, from `path` if it matches the runs, else computed and saved."""
    parsed_idx = np.array([i for i, run in enumerate(runs) if run.price is not None])
    content = MODEL_ID + PRICE_PREFILL + "".join(runs[i].prompt for i in parsed_idx)
    key = hashlib.sha256(content.encode()).hexdigest()  # catches changed prompts or model with the same parsed rows
    cached = np.load(path) if path.exists() else None
    # Caches written before the key existed are checked on run indices only
    if cached is not None and np.array_equal(cached["run_idx"], parsed_idx) and str(cached.get("key", key)) == key:
        return cached["acts"], cached["run_idx"]
    # missing, or stale because the runs, prompts or model changed since the cache was written
    acts = np.stack([get_activations(runs[i].prompt, PRICE_PREFILL) for i in parsed_idx])
    np.savez(path, acts=acts.astype(np.float16), run_idx=parsed_idx, key=key)
    return acts, parsed_idx

def get_split_accuracy_ci(correct: np.ndarray, keep: np.ndarray | None = None) -> tuple[float, float, float]:
    """Pair-bootstrap accuracy over every seed's test runs; correct is (seeds, runs), NaN for training runs."""
    tested = ~np.isnan(correct) & (True if keep is None else keep)
    return get_pair_accuracy_ci(correct[tested], np.broadcast_to(probe_pairs, correct.shape)[tested], N_BOOTSTRAP, SEED)

src/collusion_probes/market.pyLogit demand market from Fish et al. 2024, Section 2.1, with alpha = 1. Provenance: docs/prompt_provenance.md.
"""Logit demand market from Fish et al. 2024, Section 2.1, with alpha = 1. Provenance: docs/prompt_provenance.md."""

import math

import numpy as np

COST = 1.0
MAX_WTP = 4.51
NASH_PRICE = 1.47
MONOPOLY_PRICE = 1.92
DEMAND_INTERCEPT = 2.0
DIFFERENTIATION = 0.25
MARKET_SIZE = 100.0


def get_quantity(my_price: float, rival_price: float) -> float:
    mine = math.exp((DEMAND_INTERCEPT - my_price) / DIFFERENTIATION)
    rival = math.exp((DEMAND_INTERCEPT - rival_price) / DIFFERENTIATION)
    return MARKET_SIZE * mine / (mine + rival + 1)


def make_round(my_price: float, rival_price: float) -> dict[str, float]:
    quantity = get_quantity(my_price, rival_price)
    return {
        "my_price": my_price,
        "rival_price": rival_price,
        "quantity": round(quantity, 2),
        "profit": round((my_price - COST) * quantity, 2),
    }


def make_price_history(rng: np.random.Generator, n_rounds: int) -> list[dict[str, float]]:
    prices = rng.uniform(NASH_PRICE, MONOPOLY_PRICE, size=(n_rounds, 2)).round(2)
    return [make_round(float(mine), float(rival)) for mine, rival in prices]


def make_follow_history(
    rng: np.random.Generator, n_rounds: int, price_range: tuple[float, float], hold_rounds: tuple[int, int]
) -> list[dict[str, float]]:
    """C-follow: I hold each price for hold_rounds (inclusive range) rounds; the rival copies my previous price exactly."""
    low, high = price_range
    my_prices: list[float] = []
    while len(my_prices) < n_rounds:
        my_prices += [round(float(rng.uniform(low, high)), 2)] * int(rng.integers(hold_rounds[0], hold_rounds[1] + 1))
    my_prices = my_prices[:n_rounds]
    rival_prices = [round(float(rng.uniform(low, high)), 2), *my_prices[:-1]]
    return [make_round(mine, rival) for mine, rival in zip(my_prices, rival_prices)]


def shuffle_rival_prices(rng: np.random.Generator, history: list[dict[str, float]]) -> list[dict[str, float]]:
    """C-shuffle twin: same my prices and same set of rival prices, in random order, so the rival no longer follows."""
    rival_prices = rng.permutation([r["rival_price"] for r in history])
    return [make_round(r["my_price"], float(rival)) for r, rival in zip(history, rival_prices)]


def swap_roles(history: list[dict[str, float]]) -> list[dict[str, float]]:
    """C-lead twin: my and the rival's prices swapped, so I copy the rival instead of the rival copying me."""
    return [make_round(r["rival_price"], r["my_price"]) for r in history]

"""Linear probes on cached activations: one per layer, collusion (label 1) vs innocent (label 0)."""

import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline, make_pipeline
from sklearn.preprocessing import StandardScaler

PROBE_C = 1.0  # inverse L2 strength, pre-registered in DECISIONS.md
MAX_ITER = 5000  # high-dimensional fits need more than sklearn's default 100


def make_probe() -> Pipeline:
    """Standardise, then L2 logistic regression, as pre-registered."""
    return make_pipeline(StandardScaler(), LogisticRegression(C=PROBE_C, max_iter=MAX_ITER))


def get_bootstrap_ci(values: list[float] | np.ndarray, n_resamples: int, seed: int) -> tuple[float, float]:
    """95% percentile bootstrap CI of the mean, resampling values (one per pair) with replacement."""
    rng = np.random.default_rng(seed)
    boot_means = rng.choice(values, size=(n_resamples, len(values))).mean(axis=1)
    low, high = np.percentile(boot_means, [2.5, 97.5])
    return float(low), float(high)


def get_pair_accuracy_ci(
    correct: np.ndarray, pairs: np.ndarray, n_resamples: int, seed: int
) -> tuple[float, float, float]:
    """Accuracy with a 95% CI that resamples pairs, not runs: a pair's runs (and its repeats across split seeds) are one unit."""
    _, pair_of_run = np.unique(pairs, return_inverse=True)
    pair_acc = np.bincount(pair_of_run, weights=correct) / np.bincount(pair_of_run)
    low, high = get_bootstrap_ci(pair_acc, n_resamples, seed)
    return float(pair_acc.mean()), low, high


def make_text_baseline() -> Pipeline:
    """TF-IDF on the prompt text, then logistic regression: matches exact words only, not meaning."""
    return make_pipeline(TfidfVectorizer(), LogisticRegression(max_iter=MAX_ITER))

"""Run records: one pricing decision each, saved as JSON lines and grouped into pairs by history."""

import json
from dataclasses import asdict, dataclass
from pathlib import Path

NO_CUE = "none"  # condition name for the positive-control run on each history


@dataclass
class Run:
    pair_id: int  # runs sharing a pair_id share one price history
    condition: str  # "collusion", "innocent" or NO_CUE
    wording: int | None  # index into CUES[condition]; None for the no-cue run
    history: list[dict[str, float]]
    prompt: str
    completion: str
    price: float | None  # None if no price could be parsed


def load_runs(path: Path) -> list[Run]:
    return [Run(**json.loads(line)) for line in path.read_text().splitlines()]


def append_runs(path: Path, runs: list[Run]) -> None:
    """Append runs to the file, so an interrupted generation keeps every finished pair."""
    path.parent.mkdir(exist_ok=True)
    with path.open("a") as f:
        _ = f.write("".join(json.dumps(asdict(run)) + "\n" for run in runs))


def get_pair_prices(runs: list[Run]) -> dict[int, dict[str, float | None]]:
    """Price per condition for each pair, e.g. {0: {"collusion": 1.81, "innocent": 1.79, "none": 1.62}}."""
    pair_prices: dict[int, dict[str, float | None]] = {}
    for run in runs:
        if run.pair_id not in pair_prices:
            pair_prices[run.pair_id] = {}
        pair_prices[run.pair_id][run.condition] = run.price
    return pair_prices


def keep_pair(price_a: float | None, price_b: float | None, band: tuple[float, float], max_gap: float) -> bool:
    """True if both prices parsed, both sit inside the band, and they are at most max_gap apart."""
    if price_a is None or price_b is None:
        return False
    low, high = band
    both_in_band = low <= price_a <= high and low <= price_b <= high
    return both_in_band and abs(price_a - price_b) <= max_gap
